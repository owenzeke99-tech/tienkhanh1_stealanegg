local SAVE_FILE = "FiggyHub_KeyData.json"
local SAVE_DURATION = 86400

-- ==========================================
-- KEY STORAGE SYSTEM
-- ==========================================

local function saveKey()
    if type(writefile) ~= "function" then
        return false
    end

    local data = {
        key = CORRECT_KEY,
        savedAt = os.time()
    }

    local success = pcall(function()
        writefile(SAVE_FILE, HttpService:JSONEncode(data))
    end)

    return success
end

local function checkSavedKey()
    if type(isfile) ~= "function"
    or type(readfile) ~= "function" then
        return false
    end

    local success, data = pcall(function()
        if not isfile(SAVE_FILE) then
            return nil
        end

        return HttpService:JSONDecode(readfile(SAVE_FILE))
    end)

    if not success or type(data) ~= "table" then
        return false
    end

    if data.key ~= CORRECT_KEY
    or type(data.savedAt) ~= "number" then
        return false
    end

    local elapsed = os.time() - data.savedAt

    if elapsed >= 0 and elapsed < SAVE_DURATION then
        return true
    end

    if type(delfile) == "function" then
        pcall(function()
            delfile(SAVE_FILE)
        end)
    end

    return false
end

-- ==========================================
-- HUB LOADER
-- ==========================================

local function loadHub()
    local success, err = pcall(function()
        local source = game:HttpGet(LOAD_URL)
        local compiled, compileError = loadstring(source)

        assert(compiled, compileError or "Compilation failed")

        compiled()
    end)

    if not success then
        warn("Figgy Hub Load Error: " .. tostring(err))
    end

    return success
end

-- Automatically load when saved key is valid

    if loadHub() then
        return
    end
end

local function tween(object, duration, properties, style)
    local animation = TweenService:Create(
        object,
        TweenInfo.new(
            duration,
            style or Enum.EasingStyle.Quart,
            Enum.EasingDirection.Out
        ),
        properties
    )

    animation:Play()
    return animation
end

-- ==========================================
-- SHAKE ANIMATION
-- ==========================================

local isShaking = false

local function shake()

    if isShaking then return end
    isShaking = true

    local origin = MainFrame.Position

    for _, offset in ipairs({
        7, -7, 5, -5, 2, -2, 0
    }) do

        local animation = tween(MainFrame, 0.045, {
            Position = UDim2.new(
                origin.X.Scale,
                origin.X.Offset + offset,
                origin.Y.Scale,
                origin.Y.Offset
            )
        }, Enum.EasingStyle.Sine)

        animation.Completed:Wait()
    end

    MainFrame.Position = origin
    isShaking = false
end
